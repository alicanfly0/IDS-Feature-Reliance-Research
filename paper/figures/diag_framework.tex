% Figure 1: research framework (editable TikZ source; \input by main.tex).
% Three strands: the data-and-model pipeline (top row), the literature branch that fixes the
% registry and frozen groups before any ablation model is trained (middle row, left), and the
% explanation branch (SHAP, ARR; right column). SHAP never feeds group selection.
\begin{tikzpicture}[
  font=\footnotesize, >=Stealth,
  stage/.style={draw, rounded corners=2pt, fill=black!6, align=center, text width=31mm, minimum height=15mm, inner sep=2.5pt},
  lit/.style={stage, fill=white},
  expl/.style={stage, fill=black!14},
  concl/.style={stage, fill=white, very thick},
  arr/.style={->, thick},
  lab/.style={font=\footnotesize, fill=white, inner sep=1pt, align=center},
  band/.style={draw, dashed, rounded corners=4pt, inner sep=2.2mm}]
\def\dx{3.75}
% row 0: data and models
\node[stage] (d1) at (0*\dx,0) {\textbf{1 Data selection and validation}\\four tracks; checksums; row rules};
\node[stage] (d2) at (1*\dx,0) {\textbf{2 Train-only preprocessing}\\impute, drop constants, scale};
\node[stage] (d3) at (2*\dx,0) {\textbf{3 Duplicate-aware split}\\model-visible twins removed in every view};
\node[stage] (d4) at (3*\dx,0) {\textbf{4 Model training}\\AE-XGBoost r1 110, R4a-ii 70;\\DCNN 5; MES-DNN 10};
\node[expl]  (sh) at (4*\dx,0) {\textbf{Explanation}\\permutation SHAP, one AE-XGBoost model};
% row 1: literature branch, evaluation, ARR
\node[lit]   (li) at (0*\dx,-2.6) {\textbf{A Forensic literature}\\Engelen et al.; extended documentation; Lanvin et al.};
\node[lit]   (rg) at (1*\dx,-2.6) {\textbf{B Evidence registry}\\23 features, three tiers};
\node[lit]   (gr) at (2*\dx,-2.6) {\textbf{C Frozen groups}\\A--D, U; proxy closure; 10 random controls each};
\node[stage] (ev) at (3*\dx,-2.6) {\textbf{5 Evaluation}\\$\Tstar$, Monday, corrected $T'^\star$, filtered tests};
\node[expl]  (ar) at (4*\dx,-2.6) {\textbf{ARR@$K$}\\share of SHAP mass on registry features};
% row 2: inference and conclusions
\node[concl] (co) at (2*\dx,-5.2) {\textbf{8 Conclusions}\\graded by evidence type};
\node[stage] (st) at (3*\dx,-5.2) {\textbf{6 Statistics}\\paired $d_s$, sign-flip, Holm, TOST; Welch (descriptive)};
\node[expl]  (cm) at (4*\dx,-5.2) {\textbf{7 Attribution vs intervention}\\ARR, ablation effect, transfer};
% bands
\begin{scope}[on background layer]
\node[band, fit=(li)(rg)(gr), label={[font=\footnotesize\itshape, anchor=south west, align=left]north west:literature branch: members from documented defects,\\not SHAP ranks; frozen before ablation training}] {};
\node[band, fit=(sh)(ar), label={[font=\footnotesize\itshape]north:explanation branch}] {};
\end{scope}
% pipeline
\draw[arr] (d1) -- (d2); \draw[arr] (d2) -- (d3); \draw[arr] (d3) -- (d4); \draw[arr] (d4) -- (sh);
\draw[arr] (li) -- (rg); \draw[arr] (rg) -- (gr);
\draw[arr] (gr.north east) -- ([xshift=-6mm]d4.south);
\node[lab, anchor=east, fill=none] at ($(gr.north east)!0.5!([xshift=-6mm]d4.south)+(-1.5mm,0)$) {views to retrain};
\draw[arr] ([xshift=6mm]d4.south) -- ([xshift=6mm]ev.north);
\draw[arr] (ev) -- (st);
\draw[arr] (st) -- (co);
\draw[arr] (sh) -- (ar);
\draw[arr] (ar) -- (cm);
\draw[arr] (st) -- (cm);
% registry membership used by ARR: routed below the figure so it crosses no other line
\coordinate (low) at (0,-6.85);
\coordinate (rgt) at (4*\dx+2.25,0);
\draw[arr, dashed] (rg.south) -- (rg.south |- low) -- node[lab, below, pos=0.45]{registry tiers define the feature set used by ARR@$K$} (rgt |- low) -- (rgt |- ar.east) -- (ar.east);
\end{tikzpicture}
